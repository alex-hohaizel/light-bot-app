					}
				});

							dashboard: "dashboard",
							dataset: "dataset",
							proposal: "proposals",
                            organization: "organization",
							map: "maps"
						}

						const pageName = pageMatrix["dataset"];
						const url = new URL(window.location.href);

						// Встановлюємо новий шлях (route) лише якщо є відповідний запис
						if (pageName) {
							url.pathname = `/${pageName}`;
						}

						// Оновлюємо/видаляємо параметр категорії
						if (categoryCode) {
							url.searchParams.set('ext_category', categoryCode);
						} else {
							url.searchParams.delete('ext_category');
						}

						// Скидаємо пагінацію
						url.searchParams.delete('page');

					});
				}

		const updateMinimizeTooltip = () => {
			if (!minimizeButton || !backdrop) return;
			const collapsed = backdrop.classList.contains('minimized');
			const text = collapsed
				? (minimizeButton.dataset.tooltipExpand || 'Розгорнути меню')
				: (minimizeButton.dataset.tooltipCollapse || 'Згорнути меню');
			minimizeButton.setAttribute('aria-label', text);

			if (source) source.textContent = text;
		};

		updateMinimizeTooltip();

		if (minimizeButton && backdrop) {
			minimizeButton.onclick = (event) => {
				if (window.innerWidth > 1279) {
					backdrop.classList.toggle("minimized")

					backdrop.classList.remove("minimized")
					backdrop.classList.remove("expanded")
				}
			}
		}

			minimizeMobileButton.onclick = (event) => {
				backdrop.classList.remove("minimized")
				backdrop.classList.toggle("expanded")
			}
		}
	})();
