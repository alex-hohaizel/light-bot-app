    // Theme before first paint.  A saved manual choice ('light' / 'dark') always
    // wins; otherwise ('auto') the theme follows the sun in Kyiv: dark from
    // sunset to sunrise.  Sun times are computed in UTC for the date, so winter
    // and summer time need no special handling.
    window.ksTheme = (function () {
      var KEY = 'ks:theme', LAT = 50.45, LON = 30.52, R = Math.PI / 180;
      function sunUtcHours(date, rising) {
        var start = Date.UTC(date.getUTCFullYear(), 0, 0);
        var n = Math.floor((date.getTime() - start) / 864e5);
        var lngHour = LON / 15;
        var t = n + ((rising ? 6 : 18) - lngHour) / 24;
        var M = 0.9856 * t - 3.289;
        var L = (M + 1.916 * Math.sin(M * R) + 0.020 * Math.sin(2 * M * R) + 282.634 + 360) % 360;
        var RA = (Math.atan(0.91764 * Math.tan(L * R)) / R + 360) % 360;
        RA = (RA + (Math.floor(L / 90) - Math.floor(RA / 90)) * 90) / 15;
        var sinDec = 0.39782 * Math.sin(L * R), cosDec = Math.cos(Math.asin(sinDec));
        var cosH = (Math.cos(90.833 * R) - sinDec * Math.sin(LAT * R)) / (cosDec * Math.cos(LAT * R));
        var H = (rising ? 360 - Math.acos(cosH) / R : Math.acos(cosH) / R) / 15;
        return ((H + RA - 0.06571 * t - 6.622 - lngHour) % 24 + 24) % 24;
      }
      function autoTheme(now) {
        now = now || new Date();
        var h = now.getUTCHours() + now.getUTCMinutes() / 60;
        var rise = sunUtcHours(now, true), set = sunUtcHours(now, false);
        return h >= rise && h < set ? 'light' : 'dark';
      }
      function mode() {
        var m = null;
        try { m = localStorage.getItem(KEY); } catch (e) {}
        return m === 'light' || m === 'dark' ? m : 'auto';
      }
      function apply() {
        var m = mode();
        var el = document.documentElement;
        el.setAttribute('data-theme', m === 'auto' ? autoTheme() : m);
        el.setAttribute('data-theme-mode', m);
      }
      function cycle() {
        var next = { auto: 'light', light: 'dark', dark: 'auto' }[mode()];
        try { if (next === 'auto') localStorage.removeItem(KEY); else localStorage.setItem(KEY, next); } catch (e) {}
        apply();
      }
      apply();
      setInterval(function () { if (mode() === 'auto') apply(); }, 5 * 60 * 1000);
      return { cycle: cycle, autoTheme: autoTheme };
    })();

      else if (s.now === 'off') text = 'без світла' + (s.until ? ' до ' + s.until : '');
      else if (s.now === 'maybe') text = 'можливе відключення' + (s.until ? ' до ' + s.until : '');

      var cls = s && s.now === 'off' ? ' class="ks-my-off"' : '';
      return '<span' + cls + '>★ <b>' + g + '</b> — ' + text + '</span>';
    });
    box.innerHTML = lines.join('');
    box.hidden = false;
    // the strip of MY queue (not the worst of all queues), with hour marks
    var halves = {};

        bars += '<i class="' + (v === '2' ? 'mb' : '') + '" style="left:' + (i / 48 * 100) + '%;width:' + ((j - i) / 48 * 100) + '%"></i>';
